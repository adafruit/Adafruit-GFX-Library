317  [[gnu::always_inline]] void applyRotation(int16_t *x, int16_t *y) const {
318  int16_t t;
319  switch (rotation) {
320  case 1:
321  t = *x;
322  *x = WIDTH - 1 - *y;
323  *y = t;
324  break;
325  case 2:
326  *x = WIDTH - 1 - *x;
327  *y = HEIGHT - 1 - *y;
328  break;
329  case 3:
330  t = *x;
331  *x = *y;
332  *y = HEIGHT - 1 - t;
333  break;
334  }
335  }

346  void undoRotation(int16_t *x, int16_t *y) const {
347  int16_t t;
348  switch (rotation) {
349  case 1:
350  t = *x;

352  *y = WIDTH - 1 - t;
353  break;
354  case 2:

356  *y = HEIGHT - 1 - *y;
357  break;
358  case 3:
359  t = *x;
360  *x = HEIGHT - 1 - *y;
361  *y = t;
362  break;
363  }
